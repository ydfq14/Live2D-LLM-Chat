# ============================================================
# Live2D-LLM-Chat 项目依赖清单
# 安装方式: pip install -r requirements.txt
# ============================================================

# ==================== 核心依赖（本地+云端模式都需要） ====================
requests          # HTTP 请求（云端 API 调用）
numpy             # 数值计算（VAD 音频能量检测）
pygame            # 音频播放
pydub             # 音频处理（WAV/MP3 转换）
glfw              # Live2D 窗口管理
PyOpenGL          # OpenGL 绑定（代码中 import OpenGL.GL）
pyautogui         # 鼠标位置监听（眼神跟随）
live2d-py         # Live2D 模型加载与渲染
langgraph         # LangGraph 智能体引擎（StateGraph 图流转）

# ==================== LLM 模块 ====================
openai            # OpenAI SDK（支持 OpenAI / DeepSeek / MiMo 兼容接口）

# ==================== ASR 模块（本地模式需要） ====================
pyaudiowpatch     # 麦克风录音（PyAudio WASAPI 分支，代码中 import pyaudiowpatch）
keyboard          # 键盘事件监听（Ctrl 按住录音）
faster-whisper    # OpenAI Whisper 优化版（推荐，CPU/GPU 都支持）

# ==================== TTS 模块（本地模式需要） ====================
gradio            # CosyVoice WebUI 服务
gradio_client     # 连接本地 CosyVoice API
torch             # PyTorch（CosyVoice 依赖）
torchaudio        # 音频处理（CosyVoice 依赖）
librosa           # 音频加载与处理
soundfile         # 音频文件读写

# ==================== 说明 ====================
# 【云端模式】仅需安装核心依赖 + openai + langgraph
# 【本地模式】需要安装全部依赖（包括 faster-whisper / torch / gradio 等）
#
# CosyVoice 不在 PyPI 上，需单独安装：
#   git clone https://github.com/FunAudioLLM/CosyVoice.git
#   cd CosyVoice && pip install -e .
#
# Live2D 模型需从 https://github.com/Arkueid/live2d-py 获取示例模型


# ==================== LangGraph / LangChain（Agentic RAG 需要） ====================
langchain-core
langchain-openai
langgraph

# ==================== 视频模块 ====================
mediapipe 
opencv-python
chromadb
sentence-transformers
